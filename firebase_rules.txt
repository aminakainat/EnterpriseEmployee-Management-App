# Firebase Security Rules & Configuration Instructions

Follow these instructions to setup Firebase Firestore and Storage security rules for the Enterprise Employee Management App.

---

## 1. Cloud Firestore Security Rules

Paste the following rules into the **Rules** tab of your Firestore Database in the Firebase Console:

```javascript
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper function to check if the user is authenticated
    function isAuthenticated() {
      return request.auth != null;
    }
    
    // Helper function to get the current user record from the users collection
    function getUserData() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }
    
    // Helper function to check if the user has an Admin or Manager role
    function isAdminOrManager() {
      return isAuthenticated() && 
        (getUserData().role == 'admin' || getUserData().role == 'manager');
    }

    // Rules for Users collection
    match /users/{userId} {
      allow read: if isAuthenticated();
      // Only admins can modify user roles
      allow write: if isAdminOrManager();
    }

    // Rules for Employees collection
    match /employees/{employeeId} {
      allow read: if isAuthenticated();
      // Only admins can create/update/delete employees
      allow write: if isAdminOrManager();
    }

    // Rules for Attendance collection
    match /attendance/{attendanceId} {
      allow read: if isAuthenticated();
      // Employees can check in/out (create/update their own log)
      allow create, update: if isAuthenticated() && 
        (request.resource.data.employeeId == request.auth.uid || isAdminOrManager());
      // Only admins/managers can delete logs
      allow delete: if isAdminOrManager();
    }

    // Rules for Leave Requests collection
    match /leave_requests/{leaveId} {
      allow read: if isAuthenticated();
      // Employees can submit requests for themselves
      allow create: if isAuthenticated() && request.resource.data.employeeId == request.auth.uid;
      // Employees can update or delete their pending requests
      allow update, delete: if isAuthenticated() && 
        (resource.data.employeeId == request.auth.uid && resource.data.status == 'pending') || 
        isAdminOrManager();
    }

    // Rules for Notifications collection
    match /notifications/{notificationId} {
      allow read: if isAuthenticated();
      allow create: if isAdminOrManager();
      allow update: if isAuthenticated() && resource.data.employeeId == request.auth.uid;
      allow delete: if isAdminOrManager();
    }

    // Rules for Uploaded Files collection
    match /uploaded_files/{fileId} {
      allow read, write: if isAuthenticated();
    }
  }
}
```

## 2. Firebase Storage Security Rules

Paste the following rules into the **Rules** tab of your Storage Bucket in the Firebase Console:

```javascript
rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    
    function isAuthenticated() {
      return request.auth != null;
    }
    
    match /avatars/{userId}/{allPaths=**} {
      // Users can read/write their own avatar images
      allow read: if isAuthenticated();
      allow write: if isAuthenticated() && request.auth.uid == userId;
    }
    
    match /documents/{employeeId}/{allPaths=**} {
      // Employees can view and write files inside their folder, admins/managers can access everything
      allow read: if isAuthenticated();
      allow write: if isAuthenticated() && (request.auth.uid == employeeId || 
        firestore.get(/databases/(default)/documents/users/$(request.auth.uid)).data.role in ['admin', 'manager']);
    }
  }
}
```
